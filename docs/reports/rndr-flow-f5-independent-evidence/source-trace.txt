    20	    pub fn step<'a>(
    21	        &mut self,
    22	        slots: impl IntoIterator<Item = (&'a mut World, &'a mut CloseSlot, bool)>,
    23	    ) -> bool {
    24	        let mut slots: Vec<_> = slots.into_iter().collect();
    25	        for (world, close, primary) in &mut slots {
    26	            let quit = world.resource_mut::<Input>().is_some_and(|input| {
    27	                let quit = input.quit;
    28	                input.quit = false; // consume the edge; Cancel must not immediately re-arm
    29	                quit
    30	            });
    31	            self.exiting |= quit || (*primary && close.requested);
    32	        }
    33	        for (world, close, _) in &mut slots {
    34	            if (self.exiting || close.requested) && !close.active {
    35	                close.active = true;
    36	                close.needs_frame = true;
    37	                if let Some(guard) = world.resource_mut::<CloseGuard>() {
    38	                    guard.request();
    39	                }
    40	            }
    41	            close.requested = false;
    42	        }
    43	        let cancelled = slots.iter().any(|(world, close, _)| {
    44	            close.active
    45	                && world
    46	                    .resource::<CloseGuard>()
    47	                    .is_some_and(|g| matches!(g.state(), CloseState::Idle))
    48	        });
    49	        if self.exiting && cancelled {
    50	            // A cancel in any window cancels the whole process-exit transaction.
    51	            self.exiting = false;
    52	            for (world, close, _) in &mut slots {
    53	                close.active = false;
    54	                close.needs_frame = true;
    55	                if let Some(guard) = world.resource_mut::<CloseGuard>() {
    56	                    guard.cancel();
    57	                }
    58	            }
    59	            return false;
    60	        }
    61	        let ready = |world: &World| {
    62	            world
    63	                .resource::<CloseGuard>()
    64	                .is_none_or(|g| matches!(g.state(), CloseState::Ready(_)))
    65	        };
    66	        if self.exiting {
    67	            return slots.iter().all(|(world, _, _)| ready(world));
    68	        }
    69	        for (world, close, _) in &mut slots {
    70	            if close.active {
    71	                if world
    72	                    .resource::<CloseGuard>()
    73	                    .is_some_and(|g| matches!(g.state(), CloseState::Idle))
    74	                {
    75	                    close.active = false;
    76	                } else if ready(world) {
    77	                    close.closed = true;
    78	                }
    79	            }
    80	        }
    81	        false
    82	    }
